# Язык документов, создаваемых на этой доске: en или ru.
#
# Определяет только то, из какого каталога templates/<lang>/ берётся заготовка

lang: ru

# Имя проекта для внешних потребителей. Записывается в каждую находку как
# origin.project и показывает происхождение утверждения после того, как находка
# попадёт в другой проект. По умолчанию берётся имя каталога.
project: ""

# Длина случайного суффикса в локальных идентификаторах вида H-0042.k3f.
# Применяется к гипотезам, экспериментам и решениям. На находки не влияет: их
# идентификаторы уникальны глобально по построению.
#
# Значение нужно закоммитить и использовать одинаковое для всех, кто пишет на
# эту доску. Номер выдаётся локально как max+1, поэтому две ветки от одного
# коммита получат одинаковый номер 0042; различает их суффикс. Вероятность
# коллизии для k веток, претендующих на один номер, и для всей доски при k
# ветках по n документов:
#
#   длина   k=2      k=3      k=10     k=10, n=10
#   2       0.098%   0.29%    4.3%     36%
#   3       0.003%   0.009%   0.14%    1.4%
#   4       0.0001%  0.0003%  0.004%   0.04%
#
# Значение 0 отключает суффиксы. Это допустимо, только если на доску пишет один
# процесс. Работа в одиночку такой гарантии не даёт: несколько worktree, вторая
# машина или несколько агентов под одной учётной записью git пишут параллельно.
# Нельзя смешивать 0 и ненулевую длину у разных участников: тогда `H-0042`
# одновременно корректный идентификатор и префикс `H-0042.k3f`.
#
# Увеличить длину позже можно без последствий: номера не повторяются, поэтому
# старые идентификаторы без суффикса сосуществуют с новыми.
suffix_length: 3

# Число знаков в номере, дополняется нулями слева.
id_width: 4
